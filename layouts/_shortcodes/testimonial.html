" -" -}} {{- if $line -}} {{- $split := split $line " " -}} {{- $head := index $split 0 -}} {{- $rest := delimit (after 1 $split) " " -}} {{- $metrics = $metrics | append (dict "value" $head "label" $rest) -}} {{- end -}} {{- end -}} {{- end -}}
